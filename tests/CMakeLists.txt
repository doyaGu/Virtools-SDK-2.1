# The tests exercise header-only containers without the Virtools runtime.
foreach (mode IN ITEMS legacy modern)
    add_executable(ContainerTest_${mode} ContainerTest.cpp)
    target_include_directories(ContainerTest_${mode} PRIVATE "${VIRTOOLS_SDK_INCLUDE_DIR}")
    target_compile_definitions(ContainerTest_${mode} PRIVATE WIN32)
    if (mode STREQUAL "legacy")
        set_target_properties(ContainerTest_${mode} PROPERTIES CXX_STANDARD 98)
        target_compile_definitions(ContainerTest_${mode} PRIVATE VX_HAS_CXX11=0)
    else ()
        set_target_properties(ContainerTest_${mode} PROPERTIES CXX_STANDARD 11)
    endif ()
    foreach (case IN ITEMS array-growth array-reserve hash-collision hash-wraparound hash-growth)
        add_test(NAME ContainerTest_${mode}_${case} COMMAND ContainerTest_${mode} ${case})
    endforeach ()
    if (mode STREQUAL "modern")
        foreach (case IN ITEMS hash-move-constructor hash-move-assignment)
            add_test(NAME ContainerTest_${mode}_${case} COMMAND ContainerTest_${mode} ${case})
        endforeach ()
    endif ()
endforeach ()
